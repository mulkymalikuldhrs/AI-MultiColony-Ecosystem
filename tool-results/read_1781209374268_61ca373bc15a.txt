     1→"""LLM Router — Multi-Provider Failover with Cost Tracking.
     2→
     3→Provides intelligent routing across multiple LLM providers with
     4→automatic failover, health monitoring, cooldown on failure,
     5→cost tracking, and model selection (deep thinking vs quick).
     6→
     7→Features
     8→--------
     9→* Multi-provider failover (OpenAI → Anthropic → Google → NVIDIA → local)

    11→* Cooldown on failure (exponential backoff)
    12→* Cost tracking per provider and model
    13→* Model selection (deep thinking vs quick response)
    14→* Graceful fallback when providers are unavailable

    18→Uses langchain-openai, langchain-anthropic, langchain-google-genai

   183→class LLMRouter:
   184→    """Multi-provider LLM router with failover and cost tracking.
   185→
   186→    Routes LLM requests across multiple providers with automatic
   187→    failover, health monitoring, cooldown on failure, and cost tracking.
   188→
   189→    Usage::
   190→
   191→        router = LLMRouter()
   192→        router.add_provider(ProviderConfig(
   193→            provider=LLMProvider.OPENAI,
   194→            api_key="sk-...",
   195→            priority=0,
   196→        ))
   197→        response = await router.chat("Explain market volatility", tier=ModelTier.QUICK)
   198→        stats = router.get_cost_stats()
   199→    """
   200→
   201→    def __init__(self) -> None:
   202→        self._providers: Dict[LLMProvider, ProviderConfig] = {}
   203→        self._health: Dict[LLMProvider, ProviderHealth] = {}
   204→        self._cost_records: List[CostRecord] = []
   205→        self._max_consecutive_failures = 5
   206→        self._base_cooldown_seconds = 30.0
   207→
   208→    def add_provider(self, config: ProviderConfig) -> None:
   209→        """Add an LLM provider configuration.
   210→
   211→        Args:
   212→            config: ProviderConfig with provider details.
   213→        """
   214→        # Fill in default models if not specified
   215→        if not config.models:
   216→            config.models = _DEFAULT_MODELS.get(config.provider, {})
   217→        if not config.max_tokens:
   218→            config.max_tokens = _DEFAULT_MAX_TOKENS
   219→
   220→        self._providers[config.provider] = config
   221→        self._health[config.provider] = ProviderHealth(provider=config.provider)
   222→        logger.info("Added LLM provider: %s (priority=%d)", config.provider.value, config.priority)
   223→
   224→    def remove_provider(self, provider: LLMProvider) -> None:
   225→        """Remove an LLM provider.
   226→
   227→        Args:
   228→            provider: Provider to remove.
   229→        """
   230→        self._providers.pop(provider, None)
   231→        self._health.pop(provider, None)

   233→    async def chat(
   234→        self,
   235→        prompt: str,
   236→        system_prompt: Optional[str] = None,
   237→        tier: ModelTier = ModelTier.STANDARD,
   238→        max_tokens: Optional[int] = None,
   239→        preferred_provider: Optional[LLMProvider] = None,
   240→        temperature: float = 0.7,
   241→    ) -> LLMResponse:
   242→        """Send a chat request to the LLM with automatic failover.
   243→
   244→        Tries providers in priority order, falling back to the next
   245→        provider on failure. Implements cooldown on repeated failures.
   246→
   247→        Args:
   248→            prompt: User message/prompt.
   249→            system_prompt: Optional system prompt.
   250→            tier: Model tier (deep_thinking, standard, quick).
   251→            max_tokens: Maximum output tokens.
   252→            preferred_provider: Optional preferred provider to try first.
   253→            temperature: Sampling temperature.
   254→
   255→        Returns:
   256→            LLMResponse with the generated content and metadata.
   257→
   258→        Raises:
   259→            ExchangeError: If all providers fail.
   260→        """
   261→        start_time = time.monotonic()
   262→
   263→        # Build provider order
   264→        providers = self._get_provider_order(preferred_provider)
   265→
   266→        last_error = None
   267→        for provider_enum in providers:
   268→            config = self._providers.get(provider_enum)
   269→            if not config or not config.enabled:
   270→                continue
   271→
   272→            # Check cooldown
   273→            health = self._health.get(provider_enum)
   274→            if health and health.status == ProviderHealthStatus.COOLDOWN:
   275→                if health.cooldown_until:
   276→                    try:
   277→                        cooldown_time = datetime.fromisoformat(health.cooldown_until)
   278→                        if datetime.now(tz=timezone.utc) < cooldown_time:
   279→                            logger.debug("Provider %s in cooldown, skipping", provider_enum.value)
   280→                            continue
   281→                        else:
   282→                            health.status = ProviderHealthStatus.UNKNOWN
   283→                    except (ValueError, TypeError):
   284→                        pass
   285→
   286→            # Get model name
   287→            model = config.models.get(tier, "")
   288→            if not model:
   289→                continue
   290→
   291→            # Try the request
   292→            try:
   293→                content, in_tokens, out_tokens = await self._call_provider(
   294→                    config, prompt, system_prompt, model, max_tokens, temperature,
   295→                )
   296→
   297→                latency = (time.monotonic() - start_time) * 1000
   298→
   299→                # Calculate cost
   300→                cost = self._calculate_cost(provider_enum, in_tokens, out_tokens)
   301→
   302→                # Update health
   303→                self._record_success(provider_enum, latency)
   304→
   305→                # Record cost
   306→                cost_record = CostRecord(
   307→                    record_id=str(uuid.uuid4())[:8],
   308→                    provider=provider_enum,
   309→                    model=model,
   310→                    tier=tier,
   311→                    input_tokens=in_tokens,
   312→                    output_tokens=out_tokens,
   313→                    cost_usd=cost,
   314→                    latency_ms=round(latency, 2),
   315→                    success=True,
   316→                    timestamp=datetime.now(tz=timezone.utc).isoformat(),
   317→                )
   318→                self._cost_records.append(cost_record)
   319→
   320→                return LLMResponse(
   321→                    content=content,
   322→                    provider=provider_enum,
   323→                    model=model,
   324→                    tier=tier,
   325→                    input_tokens=in_tokens,
   326→                    output_tokens=out_tokens,
   327→                    cost_usd=cost,
   328→                    latency_ms=round(latency, 2),
   329→                    fallback_used=provider_enum != (preferred_provider or providers[0] if providers else None),
   330→                    timestamp=datetime.now(tz=timezone.utc).isoformat(),
   331→                )
   332→
   333→            except Exception as exc:
   334→                last_error = exc
   335→                logger.warning(
   336→                    "LLM provider %s failed: %s",
   337→                    provider_enum.value, exc,
   338→                )
   339→                self._record_failure(provider_enum)
   340→                continue
   341→
   342→        # All providers failed
   343→        raise RuntimeError(
   344→            f"All LLM providers failed. Last error: {last_error}"
   345→        )

   347→    def get_provider_health(self) -> Dict[str, ProviderHealth]:
   348→        """Get health status of all providers.
   349→
   350→        Returns:
   351→            Dict mapping provider name to health status.
   352→        """
   353→        return {p.value: h for p, h in self._health.items()}

   355→    def get_cost_stats(self) -> Dict[str, Any]:
   356→        """Get cost tracking statistics.
   357→
   358→        Returns:
   359→            Dict with cost statistics per provider and total.
   360→        """
   361→        if not self._cost_records:
   362→            return {"total_cost_usd": 0.0, "total_requests": 0, "by_provider": {}}
   363→
   364→        total_cost = sum(r.cost_usd for r in self._cost_records)
   365→        by_provider: Dict[str, Dict[str, Any]] = {}
   366→
   367→        for provider in LLMProvider:
   368→            records = [r for r in self._cost_records if r.provider == provider]
   369→            if records:
   370→                by_provider[provider.value] = {
   371→                    "total_cost_usd": round(sum(r.cost_usd for r in records), 6),
   372→                    "total_requests": len(records),
   373→                    "total_input_tokens": sum(r.input_tokens for r in records),
   374→                    "total_output_tokens": sum(r.output_tokens for r in records),
   375→                    "success_rate": round(
   376→                        sum(1 for r in records if r.success) / len(records), 4
   377→                    ),
   378→                    "avg_latency_ms": round(
   379→                        sum(r.latency_ms for r in records) / len(records), 2
   380→                    ),
   381→                }
   382→
   383→        return {
   384→            "total_cost_usd": round(total_cost, 6),
   385→            "total_requests": len(self._cost_records),
   386→            "by_provider": by_provider,
   387→        }

   389→    # ----- Internal helpers -----
   390→
   391→    def _get_provider_order(
   392→        self,
   393→        preferred: Optional[LLMProvider] = None,
   394→    ) -> List[LLMProvider]:
   395→        """Get provider order for failover routing."""
   396→        # Sort by priority (lower = higher priority)
   397→        providers = sorted(
   398→            [p for p in self._providers.keys() if self._providers[p].enabled],
   399→            key=lambda p: self._providers[p].priority,
   400→        )
   401→
   402→        # Move preferred to front if specified
   403→        if preferred and preferred in providers:
   404→            providers.remove(preferred)
   405→            providers.insert(0, preferred)
   406→
   407→        return providers

   409→    async def _call_provider(
   410→        self,
   411→        config: ProviderConfig,
   412→        prompt: str,
   413→        system_prompt: Optional[str],
   414→        model: str,
   415→        max_tokens: Optional[int],
   416→        temperature: float,
   417→    ) -> tuple[str, int, int]:
   418→        """Call a specific LLM provider.
   419→
   420→        Returns:
   421→            Tuple of (content, input_tokens, output_tokens).
   422→        """
   423→        provider = config.provider
   424→        messages = []
   425→        if system_prompt:
   426→            messages.append({"role": "system", "content": system_prompt})
   427→        messages.append({"role": "user", "content": prompt})
   428→
   429→        if provider == LLMProvider.OPENAI:
   430→            return await self._call_openai(config, messages, model, max_tokens, temperature)
   431→        elif provider == LLMProvider.ANTHROPIC:
   432→            return await self._call_anthropic(config, messages, model, max_tokens, temperature)
   433→        elif provider == LLMProvider.GOOGLE:
   434→            return await self._call_google(config, messages, model, max_tokens, temperature)
   435→        elif provider == LLMProvider.NVIDIA:
   436→            return await self._call_nvidia(config, messages, model, max_tokens, temperature)
   437→        elif provider == LLMProvider.LOCAL:
   438→            return await self._call_local(config, messages, model, max_tokens, temperature)
   439→        elif provider == LLMProvider.NVIDIA_NIM:
   440→            return await self._call_nvidia_nim(config, messages, model, max_tokens, temperature)
   441→        else:
   442→            raise ValueError(f"Unsupported provider: {provider}")

   444→    @staticmethod
   445→    async def _call_openai(
   446→        config: ProviderConfig,
   447→        messages: List[Dict[str, str]],
   448→        model: str,
   449→        max_tokens: Optional[int],
   450→        temperature: float,
   451→    ) -> tuple[str, int, int]:
   452→        """Call OpenAI API."""
   453→        try:
   454→            from langchain_openai import ChatOpenAI
   455→
   456→            llm = ChatOpenAI(
   457→                model=model,
   458→                api_key=config.api_key or "",
   459→                base_url=config.base_url,
   460→                max_tokens=max_tokens or 4096,
   461→                temperature=temperature,
   462→            )
   463→            from langchain_core.messages import HumanMessage, SystemMessage
   464→
   465→            lc_messages = []
   466→            for m in messages:
   467→                if m["role"] == "system":
   468→                    lc_messages.append(SystemMessage(content=m["content"]))
   469→                else:
   470→                    lc_messages.append(HumanMessage(content=m["content"]))
   471→
   472→            result = await llm.ainvoke(lc_messages)
   473→            content = result.content if hasattr(result, "content") else str(result)
   474→            in_tokens = result.response_metadata.get("token_usage", {}).get("prompt_tokens", 0) if hasattr(result, "response_metadata") else 0
   475→            out_tokens = result.response_metadata.get("token_usage", {}).get("completion_tokens", 0) if hasattr(result, "response_metadata") else 0
   476→            return content, in_tokens, out_tokens
   477→        except ImportError:
   478→            raise ImportError("langchain-openai is required for OpenAI provider")

   574→            lc_messages = []
   575→            for m in messages:
   576→                if m["role"] == "system":
   577→                    lc_messages.append(SystemMessage(content=m["content"]))
   578→                else:
   579→                    lc_messages.append(HumanMessage(content=m["content"]))
   580→
   581→            result = await llm.ainvoke(lc_messages)
   582→            content = result.content if hasattr(result, "content") else str(result)
   583→            in_tokens = result.response_metadata.get("token_usage", {}).get("prompt_tokens", 0) if hasattr(result, "response_metadata") else 0
   584→            out_tokens = result.response_metadata.get("token_usage", {}).get("completion_tokens", 0) if hasattr(result, "response_metadata") else 0
   585→            return content, in_tokens, out_tokens
   586→        except ImportError:
   587→            raise ImportError("langchain-openai is required for NVIDIA NIM provider")

   627→        config: ProviderConfig,
   628→        messages: List[Dict[str, str]],
   629→        model: str,
   630→        max_tokens: Optional[int],
   631→        temperature: float,

   633→        """Call NVIDIA NIM API using the NIMClient."""

   636→            from quant_nanggroe.engine.nvidia_nim.models import NIMChatMessage, NIMRole

   638→            nim_messages: list[NIMChatMessage] = []
   639→            for m in messages:
   640→                try:
   641→                    role = NIMRole(m["role"])

   643→                    role = NIMRole.USER
   644→                nim_messages.append(NIMChatMessage(role=role, content=m["content"]))
   645→
   646→            nim_config = None

   650→                    nvidia_nim_api_key=config.api_key,

   655→            try:
   656→                response = await client.chat_with_messages(
   657→                    nim_messages,
   658→                    model=model,
   659→                    temperature=temperature,
   660→                    max_tokens=max_tokens or 4096,
   661→                )
   662→                return (
   663→                    response.content,
   664→                    response.usage.prompt_tokens,
   665→                    response.usage.completion_tokens,

   674→    def _record_success(self, provider: LLMProvider, latency_ms: float) -> None:
   675→        """Record a successful provider call."""
   676→        health = self._health.get(provider)
   677→        if health:
   678→            health.status = ProviderHealthStatus.HEALTHY
   679→            health.consecutive_failures = 0
   680→            health.last_success_at = datetime.now(tz=timezone.utc).isoformat()
   681→            health.cooldown_until = None
   682→            health.total_requests += 1
   683→
   684→            # Update average latency
   685→            if health.avg_latency_ms == 0:
   686→                health.avg_latency_ms = latency_ms
   687→            else:
   688→                health.avg_latency_ms = (health.avg_latency_ms * 0.8) + (latency_ms * 0.2)
   689→
   690→            health.success_rate = (
   691→                (health.total_requests - health.total_failures) / health.total_requests
   692→                if health.total_requests > 0 else 0.0
   693→            )
   694→
   695→    def _record_failure(self, provider: LLMProvider) -> None:
   696→        """Record a failed provider call."""
   697→        health = self._health.get(provider)
   698→        if health:
   699→            health.consecutive_failures += 1
   700→            health.total_requests += 1
   701→            health.total_failures += 1
   702→            health.last_failure_at = datetime.now(tz=timezone.utc).isoformat()
   703→            health.success_rate = (
   704→                (health.total_requests - health.total_failures) / health.total_requests
   705→                if health.total_requests > 0 else 0.0
   706→            )
   707→
   708→            # Update status
   709→            if health.consecutive_failures >= self._max_consecutive_failures:
   710→                health.status = ProviderHealthStatus.UNHEALTHY
   711→            elif health.consecutive_failures >= 3:
   712→                health.status = ProviderHealthStatus.DEGRADED
   713→
   714→            # Set cooldown
   715→            if health.consecutive_failures >= 2:
   716→                cooldown_seconds = self._base_cooldown_seconds * (2 ** (health.consecutive_failures - 2))
   717→                cooldown_seconds = min(cooldown_seconds, 600)  # Max 10 minutes
   718→                from datetime import timedelta
   719→                health.cooldown_until = (
   720→                    datetime.now(tz=timezone.utc) + timedelta(seconds=cooldown_seconds)
   721→                ).isoformat()
   722→                health.status = ProviderHealthStatus.COOLDOWN

   724→    @staticmethod
   725→    def _calculate_cost(
   726→        provider: LLMProvider,
   727→        input_tokens: int,
   728→        output_tokens: int,
   729→    ) -> float:
   730→        """Calculate approximate cost for an LLM call."""
   731→        costs = _COST_PER_1K.get(provider.value, {"input": 0.0, "output": 0.0})
   732→        input_cost = (input_tokens / 1000) * costs["input"]
   733→        output_cost = (output_tokens / 1000) * costs["output"]
   734→        return round(input_cost + output_cost, 6)
