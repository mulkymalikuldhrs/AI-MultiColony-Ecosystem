   261→        Each test window should start AFTER the train window ends.
   262→        """
   263→        np.random.seed(42)
   264→        n = 400
   265→        dates = pd.date_range("2023-01-01", periods=n, freq="D")
   266→        prices = pd.DataFrame({
   267→            "SPY": 100.0 * np.cumprod(1 + np.random.normal(0.0002, 0.015, n)),
   268→        }, index=dates)
   269→        signals = pd.DataFrame({
   270→            "SPY": np.ones(n) * 0.05,
   271→        }, index=dates)
   272→
   273→        analyzer = WalkForwardAnalyzer(engine, train_window=200, test_window=50)
   274→        result = analyzer.analyze(prices, signals)
   275→
   276→        for window in result["windows"]:
   277→            # Test period should not overlap with training period
   278→            assert window.test_start >= window.train_end
