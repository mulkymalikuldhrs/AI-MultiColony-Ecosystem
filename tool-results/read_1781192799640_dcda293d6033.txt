    38→def _delay(df: pd.DataFrame, n: int) -> pd.DataFrame:
    39→    """Backward shift by n (lookahead-safe; n>=1 required)."""
    40→    if n < 1:
    41→        raise ValueError("delay requires n >= 1 (lookahead ban)")
    42→    return df.shift(n)

    45→def _ind_neutralize(x: pd.DataFrame, panel: dict) -> pd.DataFrame:
    46→    """Industry/sector neutralize: subtract the row-wise sector group mean.
    47→
    48→    If panel has a 'sector' DataFrame (same shape as close), subtract the
    49→    per-sector cross-sectional mean per row. If absent, degrade to global
    50→    cross-sectional demean (subtract row mean). This is a degraded fallback
    51→    relative to the paper's industry/subindustry neutralization; see notes.
    52→    """
    53→    sector_df = panel.get("sector")
    54→    if sector_df is None:
    55→        row_mean = x.mean(axis=1, skipna=True)
    56→        return x.sub(row_mean, axis=0)
    57→    # Per-row group demean. Iterate rows; numpy-fast enough for small panels.
    58→    arr = x.to_numpy(dtype=np.float64, na_value=np.nan).copy()
    59→    sec_arr = sector_df.to_numpy()
    60→    n_rows = arr.shape[0]
    61→    for i in range(n_rows):
    62→        row = arr[i]
    63→        sec_row = sec_arr[i]
    64→        for tag in pd.unique(sec_row):
    65→            mask = sec_row == tag
    66→            vals = row[mask]
    67→            finite = vals[~np.isnan(vals)]
    68→            if finite.size == 0:
    69→                continue
    70→            mean = finite.mean()
    71→            row[mask] = vals - mean
    72→        arr[i] = row
    73→    return pd.DataFrame(arr, index=x.index, columns=x.columns)

    81→def _rolling_prod(df: pd.DataFrame, n: int) -> pd.DataFrame:
    82→    """Rolling window product; warmup -> NaN."""
    83→    return df.rolling(window=n, min_periods=n).apply(np.prod, raw=True)

    91→def _where_ternary(cond, a, b):
    92→    """Vectorised ternary `(cond ? a : b)` returning a DataFrame.
    93→
    94→    ``cond`` is a boolean DataFrame; ``a`` / ``b`` may be DataFrame or scalar.
    95→    """
    96→    if isinstance(a, (int, float)):
    97→        a_arr = np.full_like(cond.to_numpy(dtype=np.float64), float(a))
    98→    else:
    99→        a_arr = a.to_numpy(dtype=np.float64, na_value=np.nan)
   100→    if isinstance(b, (int, float)):
   101→        b_arr = np.full_like(cond.to_numpy(dtype=np.float64), float(b))
   102→    else:
   103→        b_arr = b.to_numpy(dtype=np.float64, na_value=np.nan)
   104→    cond_arr = cond.to_numpy(dtype=bool, na_value=False) if hasattr(cond, "to_numpy") else np.asarray(cond, dtype=bool)
   105→    out = np.where(cond_arr, a_arr, b_arr)
   106→    out = np.where(np.isfinite(out), out, np.nan)
   107→    idx = cond.index if hasattr(cond, "index") else a.index
   108→    cols = cond.columns if hasattr(cond, "columns") else a.columns
   109→    return pd.DataFrame(out, index=idx, columns=cols)
