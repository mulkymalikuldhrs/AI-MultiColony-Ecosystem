   128→class TestKellyBasic:
   129→    """Basic Kelly Criterion calculations."""
   130→
   131→    def test_basic_kelly_positive_edge(self, kelly: KellyCriterion):
   132→        """With positive expectancy, Kelly should be > 0."""
   133→        params = KellyParameters(win_rate=0.6, avg_win=200.0, avg_loss=100.0)
   134→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   135→        assert result.optimal_fraction > 0
   136→
   137→    def test_basic_kelly_no_edge(self, kelly: KellyCriterion):
   138→        """With no edge (50/50, 1:1 payout), Kelly should be 0."""
   139→        params = KellyParameters(win_rate=0.5, avg_win=100.0, avg_loss=100.0)
   140→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   141→        assert result.optimal_fraction == 0.0
   142→
   143→    def test_basic_kelly_negative_edge(self, kelly: KellyCriterion):
   144→        """With negative expectancy, Kelly should be 0 (don't bet)."""
   145→        params = KellyParameters(win_rate=0.3, avg_win=100.0, avg_loss=100.0)
   146→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   147→        assert result.optimal_fraction == 0.0
   148→
   149→    def test_basic_kelly_formula(self, kelly: KellyCriterion):
   150→        """Verify Kelly formula: f* = (bp - q) / b where b = avg_win/avg_loss."""
   151→        params = KellyParameters(win_rate=0.6, avg_win=200.0, avg_loss=100.0)
   152→        # b = 200/100 = 2, p = 0.6, q = 0.4
   153→        # f* = (2*0.6 - 0.4) / 2 = 0.8 / 2 = 0.4
   154→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   155→        assert abs(result.optimal_fraction - 0.4) < 1e-6
   156→
   157→    def test_zero_avg_loss(self, kelly: KellyCriterion):
   158→        """With zero avg_loss, should return 0."""
   159→        params = KellyParameters(win_rate=0.6, avg_win=200.0, avg_loss=0.0)
   160→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   161→        assert result.optimal_fraction == 0.0
   162→
   163→    def test_high_win_rate_high_payout(self, kelly: KellyCriterion):
   164→        """Strong edge should produce larger Kelly fraction."""
   165→        params = KellyParameters(win_rate=0.8, avg_win=300.0, avg_loss=100.0)
   166→        result = kelly.calculate_kelly(params, KellyMethod.FULL_KELLY)
   167→        # b=3, p=0.8, q=0.2 → f*=(3*0.8-0.2)/3 = 2.2/3 ≈ 0.733
   168→        assert result.optimal_fraction > 0.5
