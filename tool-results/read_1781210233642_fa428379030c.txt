   131→        graph = TradingGraph(
   132→            llm_provider=provider,
   133→            deep_think_model=deep_model,
   134→            quick_think_model=quick_model,

   155→def _show_simulated_pipeline(symbols: list, trade_date: str) -> None:
   156→    """Display simulated pipeline output when LLM is not available."""
   157→    phases = [
   158→        ("Market Analysis", ["researcher", "macro", "crypto", "forex"]),
   159→        ("Signal Generation", ["strategist"]),
   160→        ("Risk Assessment", ["risk"]),
   161→        ("Portfolio Optimization", ["portfolio"]),
   162→        ("Execution Decision", ["trader"]),
   163→        ("Order Execution", ["execution"]),
   164→    ]
   165→
   166→    for phase_name, agents in phases:
   167→        console.print(f"  [bold cyan]▸ {phase_name}[/bold cyan]")
   168→        for agent in agents:
   169→            console.print(f"    [dim]→ {agent}: ready[/dim]")
   170→
   171→    console.print()
   172→    console.print(
   173→        Panel(
   174→            f"[bold green]✓ Pipeline configured for {', '.join(symbols)}[/bold green]\n"
   175→            f"[dim]Date: {trade_date} | Mode: SIMULATED[/dim]\n"
   176→            f"[dim]Configure LLM API keys for live execution[/dim]",
   177→            title="Pipeline Status",
   178→            border_style="green",
   179→        )
   180→    )

   182→
   183→def _display_pipeline_result(result: dict, symbols: list) -> None:
   184→    """Display actual pipeline execution results."""
   185→    # Status
   186→    status = "success" if not result.get("error") else "error"
   187→    status_color = "green" if status == "success" else "red"
   188→    console.print(f"[bold {status_color}]✓ Pipeline {status}[/bold {status_color}]")
   189→
   190→    # Risk verdict
   191→    verdict = result.get("risk_verdict", "VETOED")
   192→    verdict_color = "green" if verdict == "APPROVED" else "red"
   193→    console.print(f"  Risk Verdict: [bold {verdict_color}]{verdict}[/bold {verdict_color}]")
   194→
   195→    # Confidence
   196→    confidence = result.get("confidence", 0.0)
   197→    conf_color = "green" if confidence >= 0.65 else "yellow" if confidence >= 0.4 else "red"
   198→    console.print(f"  Confidence: [bold {conf_color}]{confidence:.2%}[/bold {conf_color}]")
   199→
   200→    # Decisions
   201→    decisions = result.get("decisions", [])
   202→    if decisions:
   203→        table = Table(title="Trading Decisions", box=box.ROUNDED)
   204→        table.add_column("Symbol", style="cyan")
   205→        table.add_column("Action", style="bold")
   206→        table.add_column("Confidence", style="white")
   207→        table.add_column("Reasoning", style="dim", max_width=50)
   208→        for d in decisions:
   209→            action = d.get("action", "HOLD")
   210→            action_color = "green" if action == "BUY" else "red" if action == "SELL" else "yellow"
   211→            table.add_row(
   212→                d.get("symbol", ""),
   213→                f"[{action_color}]{action}[/{action_color}]",
   214→                f"{d.get('confidence', 0):.2%}",
   215→                d.get("reasoning", "")[:50],
   216→            )
   217→        console.print(table)
   218→
   219→    # Error display
   220→    if result.get("error"):
   221→        console.print(f"  [bold red]Error: {result['error']}[/bold red]")
